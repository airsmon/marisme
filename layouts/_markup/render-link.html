{{- $destination := .Destination -}} {{- $url := urls.Parse $destination -}} {{- $siteURL := urls.Parse site.BaseURL -}} {{- $scheme := $url.Scheme | lower -}} {{- $host := $url.Hostname | lower -}} {{- $siteHost := $siteURL.Hostname | lower -}} {{- $isProtocolRelative := and (eq $scheme "") (ne $host "") -}} {{- $isWeb := or (in (slice "http" "https") $scheme) $isProtocolRelative -}} {{- $isExternal := and $isWeb (ne $host $siteHost) -}} {{- $shouldGate := .Page.IsPage -}} {{- $shouldRewrite := and $shouldGate $isExternal -}} {{- $isRelative := and (eq $scheme "") (eq $host "") -}} {{- $isMailOrTel := in (slice "mailto" "tel") $scheme -}} {{- $isAllowed := or $isWeb $isRelative $isMailOrTel -}} {{- if $isAllowed -}} {{- $href := $destination -}} {{- if $shouldRewrite -}} {{- $target := $destination -}} {{- if $isProtocolRelative -}} {{- $target = printf "%s:%s" $siteURL.Scheme $destination -}} {{- end -}} {{- $href = printf "%s?%s" ("go/" | relLangURL) (querify "url" $target) -}} {{- end -}} {{- with .Text }}{{ . }}{{ end -}} {{- if $shouldRewrite -}} ↗ （外部链接，将先进入确认页面） {{- end -}} {{- else -}} {{- with .Text }}{{ . }}{{ end -}} {{- end -}} {{- /* Chomp trailing whitespace. */ -}}
